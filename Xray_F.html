import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, GlobalAveragePooling2D, Dense, Dropout

baseline_model = Sequential([
    # First Convolution Layer
    Conv2D(32, (3,3), activation='relu', input_shape=(224,224,3)),
    # First Pooling Layer
    MaxPooling2D((2,2)),

    # Second Convolution Layer
    Conv2D(64, (3,3), activation='relu'),
    # Second Pooling Layer
    MaxPooling2D((2,2)),

    #  Low parameters, great for preventing overfitting 
    GlobalAveragePooling2D(),
    
    # Fully Connected Layer
    Dense(128, activation='relu'),

    # Prevent Overfitting
    Dropout(0.3),

    # Output Layer (2 classes for your X-ray targets)
    Dense(2, activation='softmax')
])

resnet_mod = ResNet50(
    weights='imagenet',
    include_top=False,
    input_shape=(224,224,3))

# Freeze all ResNet layers - DenseNet weights = Locked ( use them same as it is)
resnet_mod.trainable = False  # Useful for Feature Extraction

#, ResNet50 is loaded with pretrained ImageNet weights and frozen using trainable=False. 
# Therefore, the ResNet layers act only as a feature extractor. During training,
#  only the newly added Dense layers are trained on the chest X-ray dataset to classify NORMAL and PNEUMONIA.

# If, initially 'True' - Catastrophic Forgetting - Good ImageNet knowledge - Gets destroyed by small X-ray dataset

def mod_comp_fit(model, train_ds, test_ds):

    model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])

    history = model.fit(train_ds, validation_data=test_ds, epochs=15, callbacks=[early_stop])

    return history

def Acc_plot(history, title="Model Accuracy"):

    plt.plot(history.history['accuracy'])
    plt.plot(history.history['val_accuracy'])

    plt.title(title)
    plt.xlabel("Epoch")
    plt.ylabel("Accuracy")

    plt.legend(["Training", "Validation"])

    plt.show()
